Condition: Check if number is even or odd

Multiple If (Else If)

Condition: Check Positive, Negative, or Zero
if (num > 0) { console.log("Positive"); } else if (num < 0) { console.log("Negative"); } else { console.log("Zero"); }
Result will appear here...

Nested If

Condition: Check Grade from Marks
if (marks >= 0 && marks <= 100) { if (marks >= 90) { console.log("Grade A+"); } else if (marks >= 75) { console.log("Grade A"); } else if (marks >= 50) { console.log("Grade B"); } else { console.log("Fail"); } } else { console.log("Invalid Marks"); }
